+# instances can both pass the "no users yet" check and race to insert

 def seed_database(db: Session):
+    is_postgres = db.bind.dialect.name == "postgresql"
+    if is_postgres:
+        # Session-level advisory lock serialises concurrent seeders across
+        # instances; it persists across the commits below (same connection)
+        # and is released in the finally block.
+        db.execute(text("SELECT pg_advisory_lock(:k)"), {"k": _SEED_ADVISORY_LOCK_KEY})
+    try:
+        _seed_database_locked(db)
+    finally:
+        if is_postgres:
+            db.execute(text("SELECT pg_advisory_unlock(:k)"), {"k": _SEED_ADVISORY_LOCK_KEY})
+            db.commit()
