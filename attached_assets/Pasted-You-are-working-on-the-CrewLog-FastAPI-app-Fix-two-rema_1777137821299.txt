BUG 1 — iPad navigation still broken in Chrome and Safari
══════════════════════════════════════════════════════════════════════════════

The sidebar CSS approach via transform:none is unreliable because:
  a) The drawer's base CSS may use display:none (not just transform)
  b) aria-hidden="true" is set on the drawer in its initial HTML state and
     ui_nav.js never removes it on desktop — assistive trees hide the element
  c) iPad Pro 12.9" in landscape has viewport 1366px which exceeds the
     1280px sidebar threshold, triggering the sidebar — but that sidebar
     is invisible because of (a) and (b)

The correct fix is a media-query-aware JavaScript init, not CSS overrides.

─────────────────────────────────────────────────────────────────────────────
STEP 1A — Replace the sidebar CSS approach with a JS-driven desktop init
─────────────────────────────────────────────────────────────────────────────

In static/ui_nav.js, find the init() function. At the very START of init(),
before any event listeners are added, insert this desktop sidebar bootstrap:

    function init() {
        drawer = document.getElementById('nav-drawer');
        overlay = document.getElementById('nav-overlay');
        openBtn = document.getElementById('drawer-open');
        closeBtn = document.getElementById('drawer-close');
        body = document.body;

        if (!drawer) return;

        // ── Desktop sidebar: permanently open above 1279px ──────────────
        function applyLayout() {
            if (window.innerWidth >= 1280) {
                // Force sidebar visible regardless of drawer state
                drawer.style.removeProperty('display');
                drawer.setAttribute('aria-hidden', 'false');
                drawer.classList.add('drawer-open');
                body.classList.remove('drawer-scroll-lock');
                if (overlay) overlay.classList.remove('overlay-visible');
                isOpen = false;  // not in "modal open" mode on desktop
            } else {
                // Mobile/tablet: ensure sidebar is in correct closed state
                // unless it was explicitly opened by the user
                if (!isOpen) {
                    drawer.setAttribute('aria-hidden', 'true');
                    drawer.classList.remove('drawer-open');
                }
            }
        }

        applyLayout();

        // Re-apply on resize (handles orientation changes and DevTools resize)
        var resizeTimer;
        window.addEventListener('resize', function() {
            clearTimeout(resizeTimer);
            resizeTimer = setTimeout(applyLayout, 100);
        });

        if (!openBtn) { highlightActive(); return; }
        // ... (rest of init unchanged) ...

STEP 3D — Add boat stats computation to routes_boat.py
─────────────────────────────────────────────────────────────────────────────

In routes_boat.py, find the boat_setup_page GET handler. Add stats 
computation before the TemplateResponse call:

    from sqlalchemy import func
    from models import LogbookEntry, Trip as TripModel

    account_id = get_active_account_id(request)
    bp = get_or_create_boat_profile(db, account_id)

    # Compute cumulative boat stats for trips flagged use_main_boat=True
    wagmi_trips = db.query(TripModel).filter(
        TripModel.account_id == account_id,
        TripModel.use_main_boat == True
    ).all()
    wagmi_trip_ids = [t.id for t in wagmi_trips]

    # Total miles: sum of dist_day_nm across all WAGMI logbook entries
    total_nm = 0.0
    if wagmi_trip_ids:
        total_nm = float(
            db.query(func.coalesce(func.sum(LogbookEntry.dist_day_nm), 0))
            .filter(LogbookEntry.trip_id.in_(wagmi_trip_ids))
            .scalar() or 0
        )

    # Motor hours: per trip, take (max - min) eng_hours_total, sum across trips
    total_motor_h = 0.0
    for tid in wagmi_trip_ids:
        readings = db.query(LogbookEntry.eng_hours_total).filter(
            LogbookEntry.trip_id == tid,
            LogbookEntry.eng_hours_total.isnot(None)
        ).all()
        vals = [r[0] for r in readings if r[0] is not None]
        if len(vals) >= 2:
            total_motor_h += max(vals) - min(vals)

    boat_stats = {
        "total_nm":      round(total_nm, 1),
        "total_motor_h": round(total_motor_h, 1),
        "trip_count":    len(wagmi_trip_ids),
        "last_trip":     wagmi_trips[0].start_date if wagmi_trips else None,
    }

    return templates.TemplateResponse("admin_boat.html", {
        "request": request,
        "boat_profile": bp,
        "sail_profile": sp,
        "is_owner": user.is_owner,
        "boat_stats": boat_stats,          # ← add this
    })
